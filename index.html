function countUp(elId, endVal, isPct, prefix="") {

const el = document.getElementById(elId);

if(!el) return;

let duration = 3500;

let start = 0;

let startTime = null;

let end = parseFloat(endVal) || 0;

const step = (t) => {

if(!startTime) startTime = t;

let p = Math.min((t - startTime)/duration, 1);

let ease = 1 - Math.pow(1 - p, 3);

let curr = start + ease * (end - start);

el.innerText = prefix + (isPct ? curr.toFixed(1) + '%' : Math.floor(curr));

if(p < 1) requestAnimationFrame(step);

else el.innerText = prefix + (isPct ? end.toFixed(1) + '%' : end);

};

requestAnimationFrame(step);

}

function generateWallBlocks(wallObj, totalWallsAllowed) {

if(!wallObj || Object.keys(wallObj).length === 0) return "<div style='text-align:center; font-size:11px; letter-spacing: 2px; text-transform:uppercase;'>No wall data</div>";

var segments = ""; var countSoFar = 0; var totalScore = 0;

var sortedLevels = Object.keys(wallObj).map(Number).sort((a,b)=>b-a);

const wallTheme = { 15: {bg:'#4d4d4d'}, 16: {bg:'#666666'}, 17: {bg:'#808080'}, 18: {bg:'#999999'}, 19: {bg:'#b3b3b3'} };

sortedLevels.forEach(lvl => {

var count = wallObj[lvl];

if(count > 0) {

var bg = wallTheme[lvl] ? wallTheme[lvl].bg : '#333333';

var pct = ((count / totalWallsAllowed)*100).toFixed(0);

var label = (pct > 5) ? `${count}` : '';

segments += `<div class="wall-segment" style="flex: ${count}; background-color: ${bg};" title="Lvl ${lvl}: ${count} (${pct}%)"><span class="wall-label">${label}</span></div>`;

countSoFar += count;

totalScore += (lvl * count);

}

});

var remaining = totalWallsAllowed - countSoFar;

if (remaining > 0) segments += `<div class="wall-segment empty" style="flex: ${remaining}; border: 1px dashed var(--border-color);"></div>`;

let maxPossibleScore = totalWallsAllowed * 19;

let wallPct = ((totalScore / maxPossibleScore) * 100);

let displayPct = wallPct >= 100 ? "MAX" : parseFloat(wallPct.toFixed(1)) + '%';

return `

<div style="display:flex; justify-content:space-between; align-items:center; margin-bottom: 12px; font-size: 11px; color: var(--text-muted);">

<span style="letter-spacing: 2px; text-transform: uppercase;">Completion</span>

<span style="color: var(--text-main); font-size: 14px; letter-spacing: 2px; font-family: var(--font-sans);">${displayPct}</span>

</div>

<div class="wall-bar-container">${segments}</div>`;

}

function calculateGrowth(historyArr, days) {

if(historyArr.length < 2) return "0.0";

let latest = historyArr[historyArr.length-1].pct;

let latestTs = historyArr[historyArr.length-1].ts || new Date().getTime();

let targetTs = latestTs - (days * 86400000);

let pastEntry = historyArr.slice().reverse().find(h => h.ts <= targetTs) || historyArr[0];

return parseFloat((latest - pastEntry.pct).toFixed(2));

}

function filterRoster() {

const searchTerm = document.getElementById('searchInput').value.toLowerCase();

const sortBy = document.getElementById('sortSelect').value;

displayedPlayerData = globalPlayerData.filter(p => p.name.toLowerCase().includes(searchTerm));

displayedPlayerData.sort((a, b) => {

if (sortBy === 'comp') return b.percentage - a.percentage;

if (sortBy === 'alpha') return a.name.localeCompare(b.name);

if (sortBy === 'hero') {

let aH = parseInt(a.heroStats.split('/')[0]) / parseInt(a.heroStats.split('/')[1]);

let bH = parseInt(b.heroStats.split('/')[0]) / parseInt(b.heroStats.split('/')[1]);

return bH - aH;

}

if (sortBy === 'pet') {

let aP = parseInt(a.petStats.split('/')[0]) / parseInt(a.petStats.split('/')[1]);

let bP = parseInt(b.petStats.split('/')[0]) / parseInt(b.petStats.split('/')[1]);

return bP - aP;

}

if (sortBy === 'wall') {

let aW = a.walls ? Object.entries(a.walls).reduce((s, [l, c]) => s + (parseInt(l) * c), 0) : 0;

let bW = b.walls ? Object.entries(b.walls).reduce((s, [l, c]) => s + (parseInt(l) * c), 0) : 0;

return bW - aW;

}

return 0;

});

renderPlayerCards(sortBy);

}

function renderEntityBarChart(ctx, entityObj, upgradingCounts = {}) {

let style = getComputedStyle(document.body);

let sortedEntities = Object.keys(entityObj).map(name => {

return { name: name, pct: ((entityObj[name] / maxLevelsMap[name]) * 100) };

}).sort((a, b) => b.pct - a.pct);

let labels = sortedEntities.map(e => {

return upgradingCounts[e.name] ? e.name + upgradingCounts[e.name] : e.name;

});

let dataPct = sortedEntities.map(e => e.pct.toFixed(1));

let bgColors = dataPct.map(pct => parseFloat(pct) >= 100 ? '#7a8c66' : style.getPropertyValue('--chart-line').trim());

return new Chart(ctx, {

type: 'bar',

data: {

labels: labels,

datasets: [{

data: dataPct,

backgroundColor: bgColors,

borderRadius: 4

}]

},

options: {

indexAxis: 'y', responsive: true, maintainAspectRatio: false,

plugins: {

legend: { display: false },

tooltip: { enabled: false }

},

scales: {

x: { max: 100, border: { display: false }, grid: { display: false }, ticks: { display: false } },

y: { border: { display: false }, grid: { display: false }, ticks: { font: { family: 'Montserrat', size: 11 } } }

}

},

plugins: [{

id: 'barInsideLabels',

afterDatasetsDraw(chart) {

const { ctx, data } = chart;

ctx.save();

ctx.font = '300 11px "Montserrat"';

ctx.textBaseline = 'middle';

chart.getDatasetMeta(0).data.forEach((bar, index) => {

const val = data.datasets[0].data[index];

const isMaxed = parseFloat(val) >= 100;

let displayStr = isMaxed ? "MAX" : parseFloat(val).toString();

const fullTextWidth = ctx.measureText(displayStr).width;

const finalWidth = bar.getProps(['width'], true).width;

const isInside = finalWidth > fullTextWidth + 12;

let textX;

if (isInside) {

ctx.textAlign = 'right';

textX = bar.x - 8;

ctx.fillStyle = isMaxed ? '#0a0505' : '#1a1515';

if (bar.width < fullTextWidth + 12) return;

} else {

ctx.textAlign = 'left';

textX = bar.x + 8;

ctx.fillStyle = style.getPropertyValue('--text-muted').trim();

}

ctx.fillText(displayStr, textX, bar.y + 1);

});

ctx.restore();

}

}]

});

}

function renderLineChart(ctx, labels, dataPoints, labelName) {

let style = getComputedStyle(document.body);

let lineColor = style.getPropertyValue('--chart-line').trim();

let accentColor = style.getPropertyValue('--accent-main').trim();

let textColor = style.getPropertyValue('--text-muted').trim();

let grad = ctx.createLinearGradient(0, 0, 0, 200);

grad.addColorStop(0, style.getPropertyValue('--chart-fill').trim());

grad.addColorStop(1, 'rgba(255,255,255,0)');

let pointColors = dataPoints.map((_, i) => (i === 0 || i === dataPoints.length - 1) ? accentColor : lineColor);

let pointRadii = dataPoints.map((_, i) => (i === 0 || i === dataPoints.length - 1) ? 3.5 : 0);

return new Chart(ctx, {

type: 'line',

data: {

labels: labels,

datasets: [{

label: labelName, data: dataPoints, borderColor: lineColor, backgroundColor: grad,

borderWidth: 2, fill: true, tension: 0.4,

pointBackgroundColor: pointColors,

pointBorderColor: pointColors,

pointRadius: pointRadii,

pointHoverRadius: 5

}]

},

options: {

layout: { padding: { top: 35, right: 45, left: 45, bottom: 5 } },

responsive: true,

maintainAspectRatio: false,

interaction: { mode: 'index', intersect: false },

plugins: {

legend: { display: false },

tooltip: { enabled: false }

},

scales: {

x: {

display: true,

grid: { display: false, drawBorder: false },

ticks: { maxTicksLimit: 6, color: textColor, font: { size: 10, family: 'Montserrat', weight: 300, letterSpacing: 1 } }

},

y: {

display: true,

border: { display: false },

grid: { display: false },

ticks: { display: false }

}

}

},

plugins: [{

id: 'floatingLabels',

afterDatasetsDraw(chart) {

const { ctx, data } = chart;

const meta = chart.getDatasetMeta(0);

if(!meta.data || meta.data.length === 0) return;

ctx.save();

let currentStyle = getComputedStyle(document.body);

let textFill = currentStyle.getPropertyValue('--text-main').trim();

let accentColor = currentStyle.getPropertyValue('--accent-main').trim();

ctx.textBaseline = 'middle';

const firstIdx = 0;

const lastIdx = meta.data.length - 1;

const ptFirst = meta.data[firstIdx];

const ptLast = meta.data[lastIdx];

const valFirst = parseFloat(data.datasets[0].data[firstIdx]);

const valLast = parseFloat(data.datasets[0].data[lastIdx]);

ctx.font = '300 11px "Montserrat"';

ctx.textAlign = 'left';

ctx.fillStyle = textFill;

ctx.fillText(valLast + '%', ptLast.x + 8, ptLast.y);

if (lastIdx > 0) {

ctx.textAlign = 'right';

ctx.fillText(valFirst + '%', ptFirst.x - 8, ptFirst.y);

let diff = parseFloat((valLast - valFirst).toFixed(2));

if (diff !== 0) {

let diffText = diff > 0 ? `+${diff}%` : `${diff}%`;

ctx.textAlign = 'center';

ctx.fillStyle = diff > 0 ? accentColor : '#a64d4d';

ctx.font = '300 12px "Montserrat"';

let midX = (ptFirst.x + ptLast.x) / 2;

let midY = (ptFirst.y + ptLast.y) / 2 - 20;

ctx.fillText(diffText, midX, midY);

}

}

ctx.restore();

}

}]

});

}
